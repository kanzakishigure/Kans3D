cmake_minimum_required(VERSION 3.21)
project(KansPluginConfigure LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
# Match the engine's portable yaml-cpp wrapper without changing vendor sources.
set(_yaml "${KANS_SOURCE_ROOT}/Kans3D/vendor/yaml-cpp")
file(GLOB _yaml_sources "${_yaml}/src/*.cpp")
list(REMOVE_ITEM _yaml_sources "${_yaml}/src/emitterutils.cpp")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/emitterutils.cpp" "#include <cstdint>\n#include \"${_yaml}/src/emitterutils.cpp\"\n")
add_library(yaml-cpp STATIC ${_yaml_sources} "${CMAKE_CURRENT_BINARY_DIR}/emitterutils.cpp")
target_include_directories(yaml-cpp PUBLIC "${_yaml}/include")
target_compile_definitions(yaml-cpp PUBLIC YAML_CPP_STATIC_DEFINE)
add_executable(KansPluginConfigure main.cpp "${KANS_SOURCE_ROOT}/Kans3D/src/Kans3D/Plugin/PluginManifest.cpp")
target_include_directories(KansPluginConfigure PRIVATE "${KANS_SOURCE_ROOT}/Kans3D/src")
target_link_libraries(KansPluginConfigure PRIVATE yaml-cpp)
target_compile_definitions(KansPluginConfigure PRIVATE KANS_ENGINE_PLUGIN_DIR="${KANS_SOURCE_ROOT}/Plugins")
